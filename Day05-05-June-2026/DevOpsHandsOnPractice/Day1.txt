# Day 01 Basics

## Ansible task - Install Nginx on Pathnex server
---
- name: Install Nginx on Pathnex server
  hosts: all
  become: yes

  tasks: 
  - name: Install Nginx
    yum: 
      name: nginx
      state: present


## Terraform task - Create EC2 instance (c5.xlarge)

provider "aws" {
    region = "us-east-2"
}

resource "aws_instance" "pathnex" {
  ami-id = "ami-0abcd1234abcd1234"
  instance_type = "c5.xlarge"

  tags = {
    Name = "comercial-ec2"
  }
}

## Kubernetes task - Create Nginx pod
apiVersion: v1
kind: Pod
metadata: 
  name: pathnex-pod
spec:
  containers:
    - name: web
      image: nginx:latest
      ports:
        - containerPort: 80

#!/bin/bash

# Jenkins pipeline to checkout branch and list all files
pipeline{
  agent any
  stages{
    stage('Checkout'){
      steps{
        git branch: main, url: 'https://github.com/Pathnex/sample-java-app.git'
      } 
    }
    stage('list files'){
      steps{
        sh 'ls -la'
      }
    }
  }
}

## Gitlab CI git checkout
You will learn how to checkout Git repository in Gitlab CI

stages:
  - checkout

git-checkout:
  stage: checkout

    - git clone 'https://github.com/Pathnex/sample-java-app.git'

  name: pathnex-db

  serviceName: pathnex
  selectors:
    matchLabels:
      app: pathnex-db
  template:
    metadata:
      labels:
        app: pathnex-db
    
    spec:
      containers:
        - name: db
          image: mysql
          env:

## Shell script for checking service is running

#!/bin/bash

SERVICE_NAME="sshd"

if systemctl is-active --quiet $SERVICE_NAME; then
  echo "$SERVICE_NAME is active"
else
  echo "$SERVICE_NAME is in-active"
fi

# Artifacts Promotion

## Jenkins Pipeline — Promote Artifacts
You will learn how to **promote build artifacts from dev to staging**.

pipeline {
    agent any
    environment {
        INSTITUTE_NAME = "pathnex"
    }
    stages {
        stage('Checkout') {
            steps {
                git url 'https://github.com/Pathnex/sample-java-app.git'
            }
        }
        stage('Build') {
            steps {
                sh 'mvn clean package'
            }
        }
        stage('Promote') {
            steps {
                input message: "Promote artifacts to Staging environment"
                sh 'echo "Promoting artifacts for $INSTITUTE_NAME"'
            }
        }
    }
}

## GitLab CI — Artifacts Promotion
You will learn how to **promote artifacts manually using GitLab CI manual job**.

stages:
  - build
  - promote

variables:
  - INSTITUTE_NAME = "Pathnex"

build:
  stage: build
  image: maven:3.8.1-jdk-17
  script:
    - mvn clean package
  artifacts:
    paths:
      - target/*.jar
    expire_in: 1 week

promote:
  stage: promote
  image: alpine
  script: 
    - echo "Promoting to staging environment fo $INSTITUTE_NAME"
  when: manual

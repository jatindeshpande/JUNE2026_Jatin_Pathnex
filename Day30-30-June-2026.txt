## Ansible - Install Maria DB

---
 - name:Install Maria DB for Pathnex
   hosts: all
   become: yes

   tasks:
     - name: Install MariaDB
       yum:
         name: mariadb-server
         state: present
    
      - name: Start MariaDB
        service:
          name: mariadb
          state: started
          enabled: yes

## Notifications 
You will learn to send notifications on failure or success of pipeline 

pipeline {
    agent any
    environment {
        INSTITUTE_NAME = "pathnex"
    }
    stages {
        stage('Checkout') {
            steps {
                git url 'https://github.com/Pathnex/sample-java-app.git'
            }
        }
        stage('Build') {
            steps {
                sh 'mvn clean compile'
            }
        }
    }
    post {
        success {
            email to: 'team@pathnex.com'
                Subject: Pipeline succeeded
                Body: The pipeline has completed and build number is ${env.BUILD_NUMBER}
        }
        failure {
            email to: 'team@pathnex.com'
                Subject: Pipeline failed
                Body: The pipeline has failed and build number is ${env.BUILD_NUMBER}
        }
    }
}

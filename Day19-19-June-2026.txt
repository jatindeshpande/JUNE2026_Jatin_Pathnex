# Day8 - Variables, VPCs, probes, loops

## Ansible - Install packages using Variables
---
- name: Install packages using Variables
  hosts: all
  becomes: yes

  vars:
    pathnex_variables:
      - tree
      - unzip
      - vim

  tasks:
    - name: Install the required packages
      yum:
        - name: "{{ pathnex_variables }}"
          state: present

## Terraform - Create VPC and subnet
provider "aws" {
    region = "us-east-2"
}

resource "aws_vpc" "PathnexVPC" {
    cidr_block = "10.0.0.0/16"

    tags = {
        Name = "Pathnex-VPC"
    }
}

resource "aws_subnet" "Pathnexsubnet" {
    vpc_id = aws_vpc.PathnexVPC.id
    cidr_block = "10.0.1.0/24"
    map_public_ip_on_launch = true

    tags = {
        Name = "Pathnex-Subnet"
    }
}

## Kubernetes - Add Liveness and Readiness probes
apiVersion: app/v1
kind: Pod
meta-data:
  name: pathnex-probe-pod

spec:
  containers:
    - name: web
      image: nginx
      livenessProbe:
        httpGet:
          path: /
          port: 80
        initialDelaySeconds: 5
      readinessProbes:
        httpGet:
          path: /
          port: 80
        initialDelaySeconds: 3

## Jenkins Pipeline — Conditional Execution
You will learn how to **run stages based on branch or parameter conditions**.

pipeline {
  agent any
  parameters {
    string(name: 'ENVIRONMENT', default_value: 'dev', Description: 'Deployment Environment')    
  }
  environment {
    INSTITUTE_NAME = "Pathnex"
  }
  stages {
    stage('Checkout') {
      steps {
        git branch: 'main', url: 'https://github.com/Pathnex/sample-java-app.git'
      }
    }
    stage('Build') {
      steps {
        when {
          branch 'main'
        }
        steps {
          sh 'mvn clean compile'
        }
      }
    }
    stage('Deploy') {
      when {
        expression {params.ENVIRONMENT == 'dev'}
      }
      steps {
        sh 'echo "Deploying in $ENVIRONMENT by $INSTITUTE_NAME"'
      }
    }
  }
}

##  GitLab CI — Conditional Jobs
You will learn how to **run jobs based on variables or branches**.

stages:
  - build
  - deploy

variables:
  CI_ENVIRONMENT_NAME: dev
  INSTITUTE_NAME: Pathnex


build:
  stage: build
  image: jdk-17
  script: 
    - git clone https://github.com/Pathnex/sample-java-app.git
    - cd sample-java-app
    - mvn clean compile
  only:
    main

deploy:
  stage: deploy
  image: alpine:latest
  script:
    - echo 'Deploying in Environment by Pathnex'
  only:
    variables:
      - $CI_ENVIRONMENT_NAME == "dev"

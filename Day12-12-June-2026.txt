## Parallel builds
# Jenkins pipeline - Parallel Build and test

pipeline {
    agent any
    stages {
        stage('Parallel task') {
            parallel {
                stage('Build') {
                    steps {
                        sh 'mvn clean compile'
                    }
                }
                stage('Test') {
                    steps {
                        sh 'mvn test'
                    }
                }
            }
        }
    }
}

## Gitlab CI parallel jobs

## You will know how to run parallel jobs

stages:
  - build-test

maven-build:
  stage: build-test
  image: maven:3.8.1-jdk-17
  script: 
    - git clone https://github.com/Pathnex/sample-java-app.git
    - cd sample-java-app
    - mvn clean compile
  parallel: 1

maven-test: 
  stage: build-test
  image: maven:3.8.1-jdk-17
  script: 
    - cd sample-java-app
    - mvn test
  parallel: 1

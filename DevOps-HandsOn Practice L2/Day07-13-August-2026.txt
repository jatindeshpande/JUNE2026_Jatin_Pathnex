Day 07 — Automating Cloud Infrastructure with Terraform & Ansible
## Ansible — Setup Nginx with Custom Config
---
- name: Install nginx and custom config
  host: all
  become: yes

  tasks:
    - name: Install nginx
      yum:
        name: nginx
        state: present
    
    - name: Copy custom nginx.conf file
      copy:
        src: nginx.conf
        destination: /etc/nginx/nginx.conf

    - name: Start nginx server
      service:
        name: nginx
        state: started
        enabled: yes

## Terraform — Provision EC2 with IAM Role
resource "aws_iam_role" "EC2Role" {
    name = "Pathnex-Role"
    assume_role = jsonencode({
        Version = "2012-07-12"
        Statement = [
            {
                Action = "sts.Assume"
                Effect = "Allow"
                Principal = {
                    Service = "ec2.amazonaws.com"
                }
            }
        ]
    })
}

resource "aws_instance" "PathnexEC2" {
    ami = "ami-123456fddass654"
    instance_type = "m5.xlarge"
    iam_instance_profile = aws_iam_role.EC2Role.name

    tags = {
        Name = "Pathenex-Instance"
    }
}

Day 04 — Basic CI/CD Concepts with Jenkins & GitLab

## Jenkinsfile — Basic CI Pipeline

pipeline {
    agent any
    environment {
        INSTITUTE_NAME = "Pathnex"
    }
    stages {
        stage('Build') {
            step {
                sh 'mvn clean package'
            }
        }
        stage('Test') {
            steps {
                sh 'mvn test'
            }
        }
        stage('Deploy') {
            steps {
                sh 'mvn deploy'
            }
        }
    }
}

## Gitlab CI/CD basic pipeline

stages:
  - build
  - test
  - deploy

variables:
  INSTITUTE_NAME = "Pathnex"

build:
  stage: build
  image: maven:3.8.1-jdk-17
  script:
    - mvn clean package
  artifacts:
    paths:
      - target/

test:
  stage: test
  image: maven:3.8.1-jdk-17
  script:
    - mvn test

deploy:
  stage: deploy
  image: maven:3.8.1-jdk-17
  script:
    - mvn deploy

## Ansible — Install and Configure Nginx
---
- name: Install and configure Nginx
  hosts: all
  become: yes

  tasks:
    - name: Install nginx
      yum:
        name: nginx
        state: present
    
    - name: Configure nginx
      template:
        src: nginx.conf.j2
        dest: /etc/nginx/nginx.conf
    
    - name: Start the nginx service
      service: 
        name: nginx
        state: started
        enabled: yes

## Terraform — EC2 with Security Group

resource "aws_security_group" "PathnexSG" {
    name = "Pathnex-SG"

    ingress {
        from_port = 22
        to_port   = 22
        protocol  = "tcp"
        cidr_blocks = ["0.0.0.0/0"]
    }

    ingress {
        from_port = 80
        to_port   = 80
        protocol  = "tcp"
        cidr_blocks = ["0.0.0.0/0"]
    }

    egress {
        from_port = 0
        to_port   = 0
        protocol  = "-1"
        cidr_blocks = ["0.0.0.0/0"]
    }
}

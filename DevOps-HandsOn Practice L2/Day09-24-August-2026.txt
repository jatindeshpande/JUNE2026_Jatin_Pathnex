Day 09 — Docker Integration with Jenkins & GitLab
## Ansible — Setup Docker Container for Nginx
---
- name: Setup docker container for nginx
  hosts: all
  become: yes

  tasks:
    - name: Fetch nginx image
      docker_image:
        name: nginx
        source: pull

    - name: Run nginx container
      docker_container:
        name: nginx-container
        image: nginx
        state: started
        publish_ports:
          - "8080:80"

## Kubernetes — Horizontal Pod Autoscaler (HPA)
apiVersion: apps/v1
kind: Deployment
metadata:
  name: pathnex-deployment
spec: 
  replicas: 1
  selector:
    matchLabels:
      app: pathnex-app
  template:
    metadata:
      labels:
        app: pathnex-app
    spec:
      containers:
        - name: nginx
          image: nginx
          ports:
            - containerPort: 80

----------------
apiVersion: autoscaling/v2
kind: HorizontalPodAutoscaler
metadata:
  name: pathnex-hpa
spec:
  scaleTargetRef:
    apiVersion: apps/v1
    kind: Deployment
    name: pathnex-deployment
  minReplicas: 1
  maxReplicas: 10
  
  metrics:
    - type: Resource
      resource:
        name: CPU
        target: 
          type: Utilization
          averageUtilization: 50 

## Jenkinsfile — Deploy to Kubernetes
pipeline {
    agent any
    environment {
        INSTITUTE_NAME = "Pathnex"
        TEAM = "DevOps"
        ENV = "Production"
    }
    stages {
        stage('Deploy') {
            steps {
                script {
                    sh 'kubectl apply -f deployment.yaml'
                }
            }
        }
    }
}
